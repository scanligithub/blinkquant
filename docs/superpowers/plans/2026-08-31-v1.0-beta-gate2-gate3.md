### Step 1: Write the adapter

```python
"""RQAlpha adapter — runs RQAlpha and returns comparable DataFrames."""

import datetime
from dataclasses import dataclass, field
from typing import List, Optional, Dict, Any

import polars as pl


@dataclass
class RQAlphaResult:
    """RQAlpha output in BlinkQuant-compatible format."""
    trades: pl.DataFrame  # signal_date, execution_date, code, side, qty, price, fee
    equity_curve: pl.DataFrame  # date, equity, cash, positions_value
    positions_daily: pl.DataFrame  # date, code, qty, cost, market_value
    account_events: List[Dict[str, Any]] = field(default_factory=list)


def run_rqalpha(
    codes: List[str],
    start_date: datetime.date,
    end_date: datetime.date,
    initial_cash: float = 10_000_000,
    hf_repo: str = "scanli/stocka-data",
    hf_token: Optional[str] = None,
) -> RQAlphaResult:
    """Run RQAlpha backtest on given universe.

    Uses RQAlpha run_func API with a MA20 > CLOSE strategy.
    Returns RQAlphaResult with DataFrames matching BlinkQuant schema.
    """
    from rqalpha import run_func

    # Build RQAlpha config
    # Note: RQAlpha uses its own data source; for HF-backed data,
    # we need to provide a custom DataSource or use the built-in one.
    # Phase 1: Use RQAlpha's built-in data with code mapping.

    # Map BlinkQuant codes (sh.600000) to RQAlpha format (600000.XSHG)
    rqalpha_codes = [_to_rqalpha_code(c) for c in codes]

    trades_rows = []
    equity_rows = []
    positions_rows = []

    def init(context):
        context.codes = rqalpha_codes
        context.holdings = {}

    def handle_bar(context, bar_dict):
        # MA20 strategy: buy if close > MA20, sell otherwise
        import rqalpha
        from rqalpha.apis import history_bars, get_atom_data

        today = context.current_dt.date()
        target = {}

        for code in context.codes:
            bars = history_bars(code, 20, "1d", "close")
            if bars is not None and len(bars) >= 20:
                ma20 = bars.mean()
                close = bars[-1]
                if close > ma20:
                    target[code] = 1.0 / len(context.codes)

        # Record trades
        portfolio = context.portfolio
        current_positions = {p: portfolio.positions[p].total_amount for p in portfolio.positions}

        for code, weight in target.items():
            if code not in current_positions or current_positions.get(code, 0) == 0:
                # BUY
                price = bar_dict[code].close
                qty = int((portfolio.cash * weight / price) / 100) * 100
                if qty > 0:
                    trades_rows.append({
                        "signal_date": today,
                        "execution_date": today,
                        "code": _from_rqalpha_code(code),
                        "side": "BUY",
                        "qty": qty,
                        "price": price,
                        "fee": 0.0,  # RQAlpha handles fees internally
                    })

        for code in list(current_positions.keys()):
            if code not in target:
                # SELL
                if code in bar_dict:
                    price = bar_dict[code].close
                    qty = portfolio.positions[code].total_amount
                    trades_rows.append({
                        "signal_date": today,
                        "execution_date": today,
                        "code": _from_rqalpha_code(code),
                        "side": "SELL",
                        "qty": qty,
                        "price": price,
                        "fee": 0.0,
                    })

        # Record equity
        equity_rows.append({
            "date": today,
            "equity": portfolio.total_value,
            "cash": portfolio.cash,
            "positions_value": portfolio.market_value,
        })

    config = {
        "base": {
            "start_date": start_date,
            "end_date": end_date,
            "initial_cash": initial_cash,
            "benchmark": None,
            "accounts": {"stock": initial_cash},
        },
        "mod": {
            "sys_accounts": {"enabled": True},
        },
    }

    run_func(
        config=config,
        init=init,
        handle_bar=handle_bar,
    )

    # Build DataFrames
    trades_df = pl.DataFrame(trades_rows) if trades_rows else pl.DataFrame(
        schema={"signal_date": pl.Date, "execution_date": pl.Date, "code": pl.Utf8,
                "side": pl.Utf8, "qty": pl.Int64, "price": pl.Float64, "fee": pl.Float64}
    )
    equity_df = pl.DataFrame(equity_rows) if equity_rows else pl.DataFrame(
        schema={"date": pl.Date, "equity": pl.Float64, "cash": pl.Float64, "positions_value": pl.Float64}
    )

    return RQAlphaResult(
        trades=trades_df,
        equity_curve=equity_df,
        positions_daily=pl.DataFrame(),  # TODO: extract from RQAlpha
    )


def _to_rqalpha_code(code: str) -> str:
    """Convert sh.600000 → 600000.XSHG format."""
    prefix, num = code.split(".")
    exchange = {"sh": "XSHG", "sz": "XSHE", "bj": "XBJE"}.get(prefix, "XSHG")
    return f"{num}.{exchange}"


def _from_rqalpha_code(code: str) -> str:
    """Convert 600000.XSHG → sh.600000 format."""
    num, exchange = code.split(".")
    prefix = {"XSHG": "sh", "XSHE": "sz", "XBJE": "bj"}.get(exchange, "sh")
    return f"{prefix}.{num}"
```

    def _compare_trades(self) -> None:
        """Compare trade execution between BlinkQuant and RQAlpha."""
        bq_trades = self.blinkquant_result.trades if hasattr(self.blinkquant_result, 'trades') else pl.DataFrame()
        ra_trades = self.rqalpha_result.get("trades", pl.DataFrame()) if isinstance(self.rqalpha_result, dict) else pl.DataFrame()

        if bq_trades.is_empty() or ra_trades.is_empty():
            return

        # Normalize code format for comparison
        bq_norm = bq_trades.with_columns(
            pl.col("code").str.replace_all(r"^sh\.", "").str.replace_all(r"^sz\.", "").str.replace_all(r"^bj\.", "")
        )
        ra_norm = ra_trades.with_columns(
            pl.col("code").str.replace_all(r"\.\w+$", "")
        )

        # Join on (execution_date, code, side)
        joined = bq_norm.join(
            ra_norm,
            on=["execution_date", "code", "side"],
            how="outer",
            suffix="_ra",
        )

        for row in joined.iter_rows(named=True):
            date = str(row.get("execution_date", ""))
            code = row.get("code", "")
            side = row.get("side", "")

            bq_qty = row.get("qty")
            ra_qty = row.get("qty_ra")
            bq_price = row.get("price")
            ra_price = row.get("price_ra")
            bq_fee = row.get("fee")
            ra_fee = row.get("fee_ra")

            # Compare quantities
            if bq_qty is not None and ra_qty is not None and bq_qty != ra_qty:
                self.details.append(DiffDetail(
                    level="TRADE", category=DiffCategory.LOT_SIZE.value,
                    date=date, code=code, side=side,
                    blinkquant={"qty": bq_qty}, rqalpha={"qty": ra_qty},
                    diff={"qty_delta": bq_qty - ra_qty},

            # Compare prices (classify as PRICE_SEMANTIC if different)

```python
"""Long-run benchmark harness for BlinkQuant."""

import argparse
import datetime
import json
import os
import sys
import time
import tracemalloc
from pathlib import Path

_BACKEND_ROOT = str(Path(__file__).resolve().parents[1])
sys.path.insert(0, _BACKEND_ROOT)

from core.backtest_engine import BacktestEngine, TradingCalendar
from core.raw_price_store import RawPriceStore
from core.backtest_types import FeeConfig, MVP_EXECUTION_CONFIG, top_n_equal_weight_allocator
from core.fee_config import load_fee_schedule
from core.data_manager import DataManager, data_manager
from core.engine import selection_engine
from huggingface_hub import hf_hub_download
import polars as pl


KEEP_COLS = ["date", "code", "open", "high", "low", "close",
             "volume", "amount", "adjustFactor", "pctChg", "isST"]


def _log(msg):
    print(f"[{datetime.datetime.now().strftime('%H:%M:%S')}] {msg}")


def _format_bytes(b):
    for unit in ['B', 'KB', 'MB', 'GB']:
        if b < 1024:
            return f"{b:.1f} {unit}"
        b /= 1024
    return f"{b:.1f} TB"


def load_data(start_date, end_date, hf_repo, token):
    """Load HF data for given period."""
    dm = DataManager()
    history_start = start_date - datetime.timedelta(days=60)
    needed_years = list(range(history_start.year, end_date.year + 1))

    parts = []
    for year in needed_years:
        _log(f"Downloading {hf_repo}/stock_kline_{year}.parquet ...")
        p = hf_hub_download(
            repo_id=hf_repo, filename=f"stock_kline_{year}.parquet",
            repo_type="dataset", token=token,
        )
        scan = pl.scan_parquet(p)
        available = scan.collect_schema().names()
        use_cols = [c for c in KEEP_COLS if c in available]
        df = scan.select(use_cols).collect()
        df = df.filter((df["code"].hash() % dm.total_nodes) == 0)
        parts.append(df)
        _log(f"  {year}: {df.height} rows, {df['code'].n_unique()} codes")

    df = pl.concat(parts, how="diagonal")
    df = df.with_columns(pl.col("date").str.to_date("%Y-%m-%d", strict=True))
    dm.df_daily = df.sort(["code", "date"])
    dm._compute_limit_flags()
    dm._apply_forward_adjustment()
    dm._append_prev_close()
    dm._optimize_memory(dm.df_daily, "df_daily")
    dm._resample_all()
    return dm


def run_benchmark(start_date, end_signal_date, hf_repo, token, top_n=20, cash=10_000_000):
    """Run single benchmark period."""
    _log(f"Loading data for {start_date} .. {end_signal_date}")
    load_start = time.time()
    dm = load_data(start_date, end_signal_date, hf_repo, token)
    load_time = time.time() - load_start

    # Wire globals
    data_manager.df_daily = dm.df_daily
    data_manager.df_weekly = dm.df_weekly
    data_manager.df_monthly = dm.df_monthly
    data_manager.df_mapping = None
    data_manager._asof_frame_cache.clear()
    data_manager._resample_all()
    selection_engine._set_cache.clear()

    # Calendar
    calendar = TradingCalendar()
    trade_dates = sorted(
        dm.df_daily.select(pl.col("date")).unique().sort("date").to_series().to_list()
    )
    calendar.set_trade_dates(trade_dates)

    # Engine
    raw_store = RawPriceStore(hf_repo_id=hf_repo, hf_token=token)
    fee_schedule = load_fee_schedule("config/fee_schedule.yaml")
    allocator = top_n_equal_weight_allocator(top_n)

    engine = BacktestEngine(
        calendar=calendar, selection_engine=selection_engine,
        raw_price_store=raw_store, fee_config=FeeConfig(),
        execution_config=MVP_EXECUTION_CONFIG, allocator=allocator,
    )

    # Run with memory tracking
    tracemalloc.start()
    bt_start = time.time()

    result = engine.run(
        formula="CLOSE > MA(CLOSE, 20)",
        start_date=start_date,
        end_signal_date=end_signal_date,
        initial_cash=cash,
        rebalance_freq="weekly",
        fee_schedule=fee_schedule,
    )

    bt_time = time.time() - bt_start
    current, peak = tracemalloc.get_traced_memory()
    tracemalloc.stop()

    n_codes = dm.df_daily["code"].n_unique()
    n_days = len(trade_dates)

    return {
        "period": f"{start_date}..{end_signal_date}",
        "n_codes": n_codes,
        "n_trade_days": n_days,
        "n_signals": result.equity_curve.filter(pl.col("signal_date").is_not_null()).height,
        "n_trades": result.trades.height,
        "final_equity": float(result.equity_curve["equity"].tail(1)[0]) if result.equity_curve.height > 0 else 0,
        "data_load_sec": round(load_time, 2),
        "backtest_sec": round(bt_time, 2),
        "total_sec": round(load_time + bt_time, 2),
        "peak_memory": _format_bytes(peak),
        "peak_memory_bytes": peak,
    }


def main():
    parser = argparse.ArgumentParser(description="BlinkQuant Benchmark")
    parser.add_argument("--periods", type=str, default="q1,q2,q3,q4",
                        help="Comma-separated periods: q1,q2,q3,q4,1y,5y,full")
    parser.add_argument("--year", type=int, default=2024, help="Year for quarterly benchmarks")
    parser.add_argument("--hf-repo", type=str, default="scanli/stocka-data")
    parser.add_argument("--output-dir", type=str, default="tests/benchmarks")
    args = parser.parse_args()

    token = os.getenv("HF_TOKEN")
    hf_endpoint = os.getenv("HF_ENDPOINT")
    if hf_endpoint:
        os.environ["HF_ENDPOINT"] = hf_endpoint

    periods = {
        "q1": (datetime.date(args.year, 1, 2), datetime.date(args.year, 3, 29)),
        "q2": (datetime.date(args.year, 4, 1), datetime.date(args.year, 6, 28)),
        "q3": (datetime.date(args.year, 7, 1), datetime.date(args.year, 9, 30)),
        "q4": (datetime.date(args.year, 10, 1), datetime.date(args.year, 12, 31)),
        "1y": (datetime.date(args.year, 1, 2), datetime.date(args.year, 12, 31)),
        "5y": (datetime.date(args.year - 4, 1, 2), datetime.date(args.year, 12, 31)),
        "full": (datetime.date(2010, 1, 4), datetime.date(args.year, 12, 31)),
    }

    output_dir = Path(args.output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    all_results = []
    for period_name in args.periods.split(","):
        if period_name not in periods:
            _log(f"Unknown period: {period_name}, skipping")
            continue

        start, end = periods[period_name]
        _log(f"\n{'='*50}")
        _log(f"Benchmark: {period_name} ({start} .. {end})")
        _log(f"{'='*50}")

        try:
            result = run_benchmark(start, end, args.hf_repo, token)
            all_results.append(result)

            # Save individual result
            out_file = output_dir / f"benchmark_{period_name}_{args.year}.json"
            out_file.write_text(json.dumps(result, indent=2, default=str), encoding="utf-8")
            _log(f"Saved: {out_file}")

            # Print summary
            _log(f"  Codes: {result['n_codes']}")
            _log(f"  Trade days: {result['n_trade_days']}")
            _log(f"  Trades: {result['n_trades']}")
            _log(f"  Final equity: {result['final_equity']:,.2f}")
            _log(f"  Backtest time: {result['backtest_sec']}s")
            _log(f"  Peak memory: {result['peak_memory']}")
        except Exception as e:
            _log(f"ERROR: {e}")
            all_results.append({"period": period_name, "error": str(e)})

    # Save summary matrix
    summary_file = output_dir / f"benchmark_summary_{args.year}.json"
    summary_file.write_text(json.dumps(all_results, indent=2, default=str), encoding="utf-8")
    _log(f"\nSummary matrix saved: {summary_file}")


if __name__ == "__main__":
    main()
```
